# Изначально архив включался в родительский CMake-проект курса,
# который предоставлял add_catch и allocations_checker.
# Эта ветка позволяет собирать архив самостоятельно без внешних зависимостей.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.16)
    project(scheme_lessons LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    enable_testing()
    foreach(stage IN ITEMS tokenizer parser basic advanced tidy)
        set(stage_sources ${stage}/tokenizer.cpp)
        if(NOT stage STREQUAL "tokenizer")
            list(APPEND stage_sources ${stage}/parser.cpp)
        endif()
        if(NOT stage STREQUAL "tokenizer" AND NOT stage STREQUAL "parser")
            list(APPEND stage_sources ${stage}/scheme.cpp)
        endif()
        add_library(scheme_${stage} ${stage_sources})
        target_include_directories(scheme_${stage} PUBLIC
            ${CMAKE_CURRENT_SOURCE_DIR}/${stage}
            ${CMAKE_CURRENT_SOURCE_DIR}/common)
        if(NOT stage STREQUAL "parser" AND NOT stage STREQUAL "tokenizer")
            add_executable(scheme_${stage}_repl ${stage}/repl/main.cpp)
            target_link_libraries(scheme_${stage}_repl PRIVATE scheme_${stage})
        endif()
        # Оригинальные тесты компилируются через небольшой автономный runner.
        set(tests ${stage}/tests/test_tokenizer.cpp)
        if(NOT stage STREQUAL "tokenizer")
            list(APPEND tests ${stage}/tests/test_parser.cpp)
        endif()
        if(stage STREQUAL "parser")
            list(APPEND tests ${stage}/tests/test_fuzzing_1.cpp)
        endif()
        if(stage STREQUAL "basic" OR stage STREQUAL "advanced" OR stage STREQUAL "tidy")
            list(APPEND tests ${stage}/tests/test_boolean.cpp ${stage}/tests/test_integer.cpp
                ${stage}/tests/test_list.cpp ${stage}/tests/test_eval.cpp)
            list(APPEND tests ${stage}/tests/test_fuzzing_2.cpp)
        endif()
        if(stage STREQUAL "advanced" OR stage STREQUAL "tidy")
            list(APPEND tests ${stage}/tests/test_symbol.cpp ${stage}/tests/test_pair_mut.cpp
                ${stage}/tests/test_control_flow.cpp)
            list(APPEND tests ${stage}/tests/test_lambda.cpp)
        endif()
        if(stage STREQUAL "tidy")
            list(APPEND tests standalone/allocations_checker.cpp)
        endif()
        add_executable(test_scheme_${stage}_local standalone/test_main.cpp ${tests})
        target_include_directories(test_scheme_${stage}_local PRIVATE standalone)
        target_link_libraries(test_scheme_${stage}_local PRIVATE scheme_${stage})
        add_test(NAME test_${stage} COMMAND test_scheme_${stage}_local)
    endforeach()
    add_executable(scheme_show_tokens standalone/show_tokens.cpp)
    target_link_libraries(scheme_show_tokens PRIVATE scheme_tokenizer)
    add_executable(scheme_show_ast standalone/show_ast.cpp)
    target_link_libraries(scheme_show_ast PRIVATE scheme_parser)
else()
    set(SCHEME_COMMON_DIR ${CMAKE_CURRENT_SOURCE_DIR}/common)
    add_subdirectory(tokenizer)
    add_subdirectory(parser)
    add_subdirectory(basic)
    add_subdirectory(advanced)
    add_subdirectory(tidy)
endif()
